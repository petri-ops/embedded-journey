常用命令备忘

# 上传文件到板子
python -m mpremote cp D:/1.WorkDir/PLC/esp32/main.py :main.py

# 重启板子
python -m mpremote reset

# 直接在板子上运行本地脚本（不保存）
python -m mpremote run D:/1.WorkDir/PLC/esp32/seg.py

# 烧录固件（恢复出厂 MicroPython）
python -m esptool --port COM7 erase-flash
python -m esptool --port COM7 --baud 460800 write-flash 0x1000 D:/1.WorkDir/PLC/firmware/ESP32_GENERIC-20260824-v1.29.0.bin
